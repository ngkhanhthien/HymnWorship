Interface Day {
id: string; // UUID
hymnNumber: number;
date: DateTime; // enforce this to be unique
noteIds: Note[];
}

Interface Note {
id: string; // UUID
topic: string; // Use an enum to just allow this to have one of three types (MIT, Prompotings, Gratitude)

}
